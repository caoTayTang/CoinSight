\begin{frame}[fragile]{Nạp lại một ngày, kho không đổi}
\begin{columns}[T]
\column{0.6\textwidth}
\begin{lstlisting}
insert into dw.fact_ohlcv_daily as f (...)
select ... from stg_valid v
join dw.dim_asset a on a.symbol = v.symbol
on conflict (asset_key, date_key, source_key)
do update set close_price = excluded.close_price, ...
where (f.open_price, ..., f.trade_count)
  is distinct from
      (excluded.open_price, ..., excluded.trade_count)
returning asset_key, source_key;
\end{lstlisting}
\column{0.36\textwidth}
\small
Dòng chỉ được ghi khi số đo thật sự đổi. \code{RETURNING} cho biết dòng nào.\par\vspace{8pt}
Mỗi lần nạp phải thỏa\\[2pt]
\code{staged = rejected + duplicate + accepted}\par\vspace{8pt}
\color{muted}Checksum SHA-256 của bảng fact trước và sau khi nạp lại bằng nhau.
\end{columns}
\end{frame}

\begin{frame}[fragile]{Trả lời câu hỏi mở đầu bằng CUBE}
\begin{columns}[T]
\column{0.6\textwidth}
\begin{lstlisting}
select case when grouping(category) = 1
            then 'ALL' else category end,
       year, quarter,
       avg(daily_return), stddev_samp(daily_return)
from mart.v_daily_return
join dw.dim_date using (date_key)
where not is_stablecoin
group by cube (category, year, quarter);
\end{lstlisting}
\column{0.37\textwidth}
\scriptsize
\begin{tabular}{@{}ll@{}}
\code{ROLLUP} & tháng $\rightarrow$ quý $\rightarrow$ năm \\
\code{CUBE} & 8 tổ hợp nhóm × năm × quý \\
\code{GROUPING SETS} & 6 tổ hợp coin, phiên, giờ \\
\code{rank()} & top 3 tăng, giảm mỗi quý \\
\end{tabular}\par\vspace{8pt}
\color{muted}Materialized view, refresh trong cùng transaction với lần nạp. Dòng tổng ghi \code{'ALL'}, không để \code{NULL}.

Đặc trưng: \code{return\_1d}, \code{return\_7d}, \code{range\_pct}, \code{volume\_change}, \code{hourly\_range} (từ đủ 24 nến giờ).\par\vspace{6pt}
Logistic regression; 30 ngày kiểm định, 30 ngày kiểm tra, chia theo thời gian.
\column{0.44\textwidth}
\card{0.86\linewidth}{\kicker{Cổng công bố}\small
$BS = \frac{1}{N}\sum (p_i - y_i)^2$\\[4pt]
Brier của mô hình phải thấp hơn baseline trên cả hai tập.\\[4pt]
\color{muted}Không đạt thì giao diện ghi ``Model below baseline. Signal withheld.''}
